## Component
```tsx
"use client";

// X-Ray Figma Cursor — Goodmotion
// Website: https://goodmotion.vercel.app/
// Dependencies: npm i gsap @gsap/react

import React, { useEffect, useRef, useState } from "react";
import { createPortal } from "react-dom";
import gsap from "gsap";
import { useGSAP } from "@gsap/react";

export interface XRayFigmaCursorProps {
    /** Label text displayed inside the multiplayer pill tag */
    name?: string;
    /** Size of the Figma pointer arrow in pixels */
    size?: number;
    /** Lerp smoothing factor or GSAP ease equation */
    ease?: number | string;
    /** Duration in seconds for pointer follow animation */
    duration?: number;
    /** Hide the native OS/browser mouse pointer */
    hideDefaultCursor?: boolean;
    /** Optional custom z-index for the cursor layer */
    zIndex?: number;
    /** Optional additional CSS class name */
    className?: string;
    /** Optional inline style overrides */
    style?: React.CSSProperties;
    /** Optional children to render alongside the cursor */
    children?: React.ReactNode;
}

export function XRayFigmaCursor({
    name = "you",
    size = 22,
    ease = 0.34,
    duration,
    hideDefaultCursor = true,
    zIndex = 999999,
    className,
    style,
    children,
}: XRayFigmaCursorProps) {
    const [mounted, setMounted] = useState(false);
    const [isTouchDevice, setIsTouchDevice] = useState(false);
    const [isVisible, setIsVisible] = useState(false);

    const cursorRef = useRef<HTMLDivElement>(null);

    // Initial environment and touch capability detection
    useEffect(() => {
        if (typeof window !== "undefined") {
            const hasFinePointer = window.matchMedia("(pointer: fine)").matches;
            const isCoarse = window.matchMedia("(pointer: coarse)").matches;
            const hasTouch =
                typeof navigator !== "undefined" && navigator.maxTouchPoints > 0;
            const isTouchOnly = !hasFinePointer && (isCoarse || hasTouch);

            setIsTouchDevice(isTouchOnly);
        }
        setMounted(true);
    }, []);

    // Safely inject and cleanup global cursor-hiding style sheet
    useEffect(() => {
        if (!mounted || isTouchDevice || !hideDefaultCursor) return;

        const styleId = "goodmotion-xray-cursor-hide";
        let styleEl = document.getElementById(styleId) as HTMLStyleElement | null;
        let isOwner = false;

        if (!styleEl) {
            styleEl = document.createElement("style");
            styleEl.id = styleId;
            styleEl.textContent = `
                html, body, *, *::before, *::after {
                    cursor: none !important;
                }
            `;
            document.head.appendChild(styleEl);
            isOwner = true;
        }

        return () => {
            if (isOwner && styleEl && styleEl.parentNode) {
                styleEl.parentNode.removeChild(styleEl);
            }
        };
    }, [mounted, isTouchDevice, hideDefaultCursor]);

    // Calculate animation parameters
    const animDuration =
        typeof duration === "number"
            ? duration
            : typeof ease === "number"
            ? ease
            : 0.25;

    const animEase = typeof ease === "string" ? ease : "power3.out";

    // GSAP pointer tracking with automatic ticker sleep and cleanup
    useGSAP(
        () => {
            if (!mounted || isTouchDevice || !cursorRef.current) return;

            const xTo = gsap.quickTo(cursorRef.current, "x", {
                duration: animDuration,
                ease: animEase,
            });
            const yTo = gsap.quickTo(cursorRef.current, "y", {
                duration: animDuration,
                ease: animEase,
            });

            let hasPositioned = false;

            const handlePointerMove = (e: PointerEvent) => {
                // Safeguard against touch inputs on hybrid/touchscreen laptops
                if (e.pointerType === "touch") return;

                if (!hasPositioned) {
                    // Position instantly on the first mouse event to eliminate jump artifacts
                    gsap.set(cursorRef.current, { x: e.clientX, y: e.clientY });
                    hasPositioned = true;
                    setIsVisible(true);
                } else {
                    xTo(e.clientX);
                    yTo(e.clientY);
                    setIsVisible((prev) => (!prev ? true : prev));
                }
            };

            const handlePointerLeave = () => {
                setIsVisible(false);
            };

            window.addEventListener("pointermove", handlePointerMove, {
                passive: true,
            });
            document.addEventListener("mouseleave", handlePointerLeave);

            return () => {
                window.removeEventListener("pointermove", handlePointerMove);
                document.removeEventListener("mouseleave", handlePointerLeave);
            };
        },
        {
            scope: cursorRef,
            dependencies: [mounted, isTouchDevice, animDuration, animEase],
        }
    );

    if (!mounted || isTouchDevice || typeof document === "undefined") {
        return null;
    }

    return createPortal(
        <div
            ref={cursorRef}
            aria-hidden="true"
            className={className}
            style={{
                position: "fixed",
                top: 0,
                left: 0,
                zIndex,
                pointerEvents: "none",
                mixBlendMode: "difference",
                opacity: isVisible ? 1 : 0,
                transition: "opacity 0.15s ease",
                willChange: "transform",
                display: "inline-flex",
                alignItems: "flex-start",
                ...style,
            }}
        >
            {/* Figma Pointer Arrow */}
            <svg
                width={size}
                height={size}
                viewBox="0 0 24 24"
                style={{
                    display: "block",
                    flexShrink: 0,
                    overflow: "visible",
                }}
            >
                <path
                    d="M5 2l14 7-6 1.7L11 18z"
                    fill="#FFFFFF"
                    stroke="#FFFFFF"
                    strokeWidth={3.5}
                    strokeLinejoin="round"
                    strokeLinecap="round"
                />
            </svg>

            {/* Multiplayer Tag (~6px offset from arrow) */}
            {name && (
                <span
                    style={{
                        marginLeft: 1,
                        marginTop: Math.round(size * 0.55),
                        backgroundColor: "#FFFFFF",
                        color: "#000000",
                        fontSize: 12,
                        fontWeight: 600,
                        lineHeight: 1.2,
                        padding: "4px 10px",
                        borderRadius: 999,
                        whiteSpace: "nowrap",
                        fontFamily:
                            "Inter, -apple-system, BlinkMacSystemFont, sans-serif",
                        letterSpacing: "-0.02em",
                        userSelect: "none",
                    }}
                >
                    {name}
                </span>
            )}

            {children}
        </div>,
        document.body
    );
}

export default XRayFigmaCursor;

```
